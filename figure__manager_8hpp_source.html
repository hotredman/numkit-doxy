30// Count top-level elements in a flat JSON number array "[a,b,c]" cheaply
31// (commas + 1) — gates the expensive parse on series size.
32inline std::size_t countFlatElements(const std::string &j) {
33 std::size_t commas = 0;
34 bool any = false;
35 for (char c : j) {
36 if (c == ',') commas++;
37 else if (c != '[' && c != ']' && c != ' ' && c != '\t' && c != '\n') any = true;
38 }
39 return any ? commas + 1 : 0;
40}

42// Parse a flat JSON number array → doubles. "null" → NaN; bare/quoted
43// Inf/NaN handled by strtod. The figure xJson is engine-built (well-formed).
44inline std::vector<double> parseFlatDoubles(const std::string &j) {
45 std::vector<double> out;
46 const char *p = j.c_str();
47 const char *end = p + j.size();
48 while (p < end && *p != '[') ++p;
49 if (p < end) ++p; // skip '['
50 while (p < end) {
51 while (p < end && (*p == ' ' || *p == ',' || *p == '"' || *p == '\t' || *p == '\n')) ++p;
52 if (p >= end || *p == ']') break;
53 if ((p[0] == 'n' || p[0] == 'N') && (p + 4 <= end) &&
54 (p[1] == 'u') && (p[2] == 'l') && (p[3] == 'l')) {
55 out.push_back(std::nan(""));
56 p += 4;
57 continue;
58 }
59 char *q = nullptr;
60 double v = std::strtod(p, &q);
61 if (q == p) { ++p; continue; } // skip an unparseable token
62 out.push_back(v);
63 p = q;
64 }
65 return out;
66}

68// Serialize doubles → flat JSON array. NaN → null, ±Inf → "Inf"/"-Inf".
69inline std::string serializeFlatDoubles(const std::vector<double> &v) {
70 std::ostringstream os;
71 os << "[";
72 os.precision(17);
73 for (std::size_t i = 0; i < v.size(); ++i) {
74 if (i) os << ",";
75 double x = v[i];
76 if (std::isnan(x)) os << "null";
77 else if (std::isinf(x)) os << (x > 0 ? "\"Inf\"" : "\"-Inf\"");
78 else os << x;
79 }
80 os << "]";
81 return os.str();
82}

546 void deleteHandle(int id)
547 {
548 auto it = handleRegistry_.find(id);
549 if (it == handleRegistry_.end() || it->second.deleted)
550 return;
551 it->second.deleted = true;
552 if (!it->second.isChart())
553 return;
554 // Remove the dataset from its axes so the render no longer
555 // carries it; reindex the surviving chart handles of that axes
556 // (their dataset slots shift down by one).
557 DatasetInfo *ds = datasetOf(it->second);
558 if (!ds)
559 return;
560 auto fit = figures_.find(it->second.figureId);
561 if (fit == figures_.end() || it->second.axesIndex >= fit->second.axes.size())
562 return;
563 auto &ax = fit->second.axes[it->second.axesIndex];
564 if (it->second.datasetIndex >= ax.datasets.size())
565 return;
566 ax.datasets.erase(ax.datasets.begin() + it->second.datasetIndex);
567 fit->second.modified = true;
568 for (auto &[hid, rec] : handleRegistry_) {
569 if (!rec.deleted && rec.figureId == it->second.figureId
570 && rec.axesIndex == it->second.axesIndex
571 && rec.datasetIndex > it->second.datasetIndex)
572 --rec.datasetIndex;
573 }

689 // Large line-series → downsampled-preview thresholds. Below the
690 // threshold the full array is shipped and the JS-side viewport
691 // decimation (Phase 1) keeps full zoom detail; at/above it the engine
692 // ships a small preview + retains raw x/y for getSeriesDisplayTile.
693 static constexpr std::size_t kSeriesPreviewThreshold = 1000000; // points
694 static constexpr int kSeriesPreviewCols = 2000; // preview cols

696 void pushDataset(DatasetInfo ds)
697 {
698 auto &ax = currentAxes();

700 ds.yside = ax.activeYside;

705 // For a huge line/stairs series with ascending x, retain raw x/y for
706 // viewport tiles and replace the inline JSON with a small M4 preview —
707 // the frontend then receives O(previewCols) points, not O(N). Mirrors
708 // imagesc's downsampled-preview path. No-op for small or non-monotonic
709 // series (they keep their full inline JSON + the JS-side decimation).
710 static void maybeDownsampleSeries(DatasetInfo &ds)
711 {
712 if (ds.type != "line" && ds.type != "stairs") return;
713 if (ds.xJson.empty() || ds.yJson.empty()) return;
714 if (figdetail::countFlatElements(ds.xJson) <= kSeriesPreviewThreshold) return;
715 auto xr = figdetail::parseFlatDoubles(ds.xJson);
716 auto yr = figdetail::parseFlatDoubles(ds.yJson);
717 const std::size_t n = xr.size();
718 if (n != yr.size() || n <= kSeriesPreviewThreshold) return;
719 // Decimation buckets by x — valid only for ascending x. Bail on a
720 // non-monotonic series (rare for huge data; keeps the full JSON).
721 for (std::size_t i = 1; i < n; ++i) if (xr[i] < xr[i - 1]) return;
722 double xlo = xr.front(), xhi = xr.back();
723 if (xhi < xlo) std::swap(xlo, xhi);
724 double ylo = INFINITY, yhi = -INFINITY;
725 for (double v : yr) if (std::isfinite(v)) { ylo = std::min(ylo, v); yhi = std::max(yhi, v); }
726 if (!std::isfinite(ylo)) { ylo = 0; yhi = 0; }
727 auto prev = decimateM4(xr.data(), yr.data(), n, xlo, xhi, kSeriesPreviewCols);
728 ds.xJson = figdetail::serializeFlatDoubles(prev.x);
729 ds.yJson = figdetail::serializeFlatDoubles(prev.y);
730 ds.xRaw = std::move(xr);
731 ds.yRaw = std::move(yr);
732 ds.seriesDownsampled = true;
733 ds.seriesN = n;
734 ds.sxLo = xlo; ds.sxHi = xhi; ds.syLo = ylo; ds.syHi = yhi;
735 }

1184 // to `width` columns. algo: 0=M4 (default), 1=LTTB, 2=none. Empty when
1185 // the dataset isn't a downsampled series or indices are out of range —
1186 // the frontend refines zoom detail without ever shipping the full array.
1187 DecimatedSeries getSeriesDisplayTile(int figId, int axIdx, int dsIdx,
1188 double x0, double x1, int width, int algo) const
1189 {

1191 auto it = figures_.find(figId);
1192 if (it == figures_.end()) return out;
1193 const auto &fig = it->second;

1229 const size_t fullCols = ds.originalCols;
1230 if (fullRows == 0 || fullCols == 0) return false;
1231 if (ds.zQuantized.size() < fullRows * fullCols) return false;
1232
1233 // Clamp source-rect to the dataset bounds.
1234 if (srcR0 < 0) { srcH += srcR0; srcR0 = 0; }
1235 if (srcC0 < 0) { srcW += srcC0; srcC0 = 0; }
1236 const double srcR1 = std::min(static_cast<double>(fullRows), srcR0 + srcH);
1237 const double srcC1 = std::min(static_cast<double>(fullCols), srcC0 + srcW);
1238 if (srcR1 <= srcR0 || srcC1 <= srcC0) return false;
1239
1240 // Log axes need a strictly-positive lower bound.
1241 if (yLog && srcR0 <= 0) return false;
1242 if (xLog && srcC0 <= 0) return false;
1243
1244 // ── LOD pyramid: pick the smallest level whose cells are ≤1 per
1245 // display-pixel. At level k each cell covers 2^k × 2^k of L0, so
1246 // the optimal level is floor(log2(min(rStep, cStep))) clamped to
1247 // [0, 10]. Building higher levels lazily — first zoom-out builds
1248 // up the tree once, subsequent zooms reuse cached levels.
1249 const double rStep0 = (srcR1 - srcR0) / displayH;
1250 const double cStep0 = (srcC1 - srcC0) / displayW;
1251 const double minStep = std::min(rStep0, cStep0);
1252 int level = 0;
1253 if (minStep > 1.5) {
1254 level = static_cast<int>(std::floor(std::log2(minStep)));
1255 if (level < 0) level = 0;
1256 if (level > 10) level = 10;
1257 }
1258 ensureLOD(ds, level);
1259
1260 // Resolve the chosen level's data + dims.
1261 const uint8_t *lvlData;
1262 size_t lvlRows, lvlCols;
1263 if (level == 0) {
1264 lvlData = ds.zQuantized.data();
1265 lvlRows = fullRows;
1266 lvlCols = fullCols;
1267 } else if (level <= static_cast<int>(ds.lodLevels.size())) {
1268 lvlData = ds.lodLevels[level - 1].data();
1269 lvlRows = ds.lodDims[level - 1].first;
1270 lvlCols = ds.lodDims[level - 1].second;
1271 } else {
1272 // ensureLOD couldn't build (e.g. dataset too small) — fall back to L0.
1273 lvlData = ds.zQuantized.data();
1274 lvlRows = fullRows;
1275 lvlCols = fullCols;
1276 level = 0;
1277 }
1278
1279 // Map source-rect from L0 coords to the chosen level's coords.
1280 const double scaleFactor = static_cast<double>(1 << level);
1281 const double lvlR0 = srcR0 / scaleFactor;
1282 const double lvlC0 = srcC0 / scaleFactor;
1283 const double lvlR1 = srcR1 / scaleFactor;
1284 const double lvlC1 = srcC1 / scaleFactor;
1285 const double rStep = (lvlR1 - lvlR0) / displayH;
1286 const double cStep = (lvlC1 - lvlC0) / displayW;
1287 const int rBlock = std::max(1, static_cast<int>(std::ceil(rStep)));
1288 const int cBlock = std::max(1, static_cast<int>(std::ceil(cStep)));
1289
1290 // Log-ratios computed in level coords (log preserves ratios).
1291 const double yLogRatio = yLog ? std::log(lvlR1 / lvlR0) : 0.0;
1292 const double xLogRatio = xLog ? std::log(lvlC1 / lvlC0) : 0.0;
1293
1294 for (int orow = 0; orow < displayH; ++orow) {
1295 const double v = (orow + 0.5) / displayH;
1296 const double rCenter = yLog
1297 ? lvlR0 * std::exp(v * yLogRatio)
1298 : lvlR0 + v * (lvlR1 - lvlR0);
1299 const int rA = std::max(0, static_cast<int>(rCenter) - rBlock / 2);
1300 const int rB = std::min(static_cast<int>(lvlRows), rA + rBlock);
1301 for (int ocol = 0; ocol < displayW; ++ocol) {
1302 const double u = (ocol + 0.5) / displayW;
1303 const double cCenter = xLog
1304 ? lvlC0 * std::exp(u * xLogRatio)
1305 : lvlC0 + u * (lvlC1 - lvlC0);
1306 const int cA = std::max(0, static_cast<int>(cCenter) - cBlock / 2);
1307 const int cB = std::min(static_cast<int>(lvlCols), cA + cBlock);
1308 int sum = 0, n = 0;
1309 for (int cc = cA; cc < cB; ++cc) {
1310 for (int rr = rA; rr < rB; ++rr) {
1311 const uint8_t q = lvlData[cc * lvlRows + rr];
1312 if (q != 255) { sum += q; ++n; }
1313 }
1314 }
1315 out[orow * displayW + ocol] = (n > 0)
1316 ? static_cast<uint8_t>((sum + n / 2) / n)
1317 : uint8_t{255};
1318 }
1319 }
1320 return true;
1321 }

1331 void ensureLOD(const DatasetInfo &ds, int targetLevel) const
1332 {
1333 while (static_cast<int>(ds.lodLevels.size()) < targetLevel) {
1334 size_t pRows, pCols;
1335 const uint8_t *parent;
1336 if (ds.lodLevels.empty()) {
1337 parent = ds.zQuantized.data();
1338 pRows = ds.originalRows;
1339 pCols = ds.originalCols;
1340 } else {

1343 pCols = dims.second;
1344 parent = ds.lodLevels.back().data();
1345 }
1346 const size_t cRows = (pRows + 1) / 2;
1347 const size_t cCols = (pCols + 1) / 2;
1348 if (cRows < 32 || cCols < 32) break; // stop refining
1349
1350 std::vector<uint8_t> child(cRows * cCols);
1351 for (size_t cc = 0; cc < cCols; ++cc) {
1352 const size_t pc0 = 2 * cc;
1353 const size_t pc1 = std::min(pCols, pc0 + 2);
1354 for (size_t rr = 0; rr < cRows; ++rr) {
1355 const size_t pr0 = 2 * rr;
1356 const size_t pr1 = std::min(pRows, pr0 + 2);
1357 int sum = 0, n = 0;
1358 for (size_t pc = pc0; pc < pc1; ++pc) {
1359 for (size_t pr = pr0; pr < pr1; ++pr) {
1360 const uint8_t q = parent[pc * pRows + pr];
1361 if (q != 255) { sum += q; ++n; }
1362 }
1363 }
1364 child[cc * cRows + rr] = (n > 0)
1365 ? static_cast<uint8_t>((sum + n / 2) / n)
1366 : uint8_t{255};
1367 }
1368 }
1369 ds.lodLevels.push_back(std::move(child));
1370 ds.lodDims.emplace_back(cRows, cCols);
1371 }
