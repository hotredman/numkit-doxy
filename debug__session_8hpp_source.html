3// DebugSession: owns compilation + VM state for pausable debug execution.
4// Replaces the WASM replay hack with a clean start/resume/snapshot API.
5//
6// Usage:
7// DebugSession session(engine);
8// session.setBreakpoints({5, 10});
9// auto status = session.start("x = 1;\ny = 2;\n");
10// if (status == ExecStatus::Paused) {
11// auto snap = session.snapshot();
12// // show snap.line, snap.variables to the user
13// status = session.resume(DebugAction::Continue);
14// }
15//

57 // ── dbstop if error ──────────────────────────────────────
58 // When enabled, an uncaught error during the run pauses at the failing
59 // line (frames intact) instead of ending the session, so state can be
60 // inspected. Resuming lets the error finally propagate (session ends with
61 // it). Set before start().
62 void setStopOnError(bool enabled);

66 // ── Frame navigation (dbup / dbdown) ─────────────────────
67 // Move the inspection focus one frame toward the base (frameUp / dbup) or
68 // back toward the current frame (frameDown / dbdown), WITHOUT resuming.
69 // snapshot() and eval() that follow operate on the newly selected frame —
70 // so a caller's locals can be inspected and edited at a pause inside a
71 // nested call. Returns true if the focus moved (false at the stack ends).
72 // The focus resets to the deepest frame on every pause.
73 bool frameUp();
74 bool frameDown();
75 // Current focus depth (0 = deepest/current frame) and the live frame count.
76 size_t selectedFrame() const { return selectedDepth_; }
77 size_t frameCount() const;

109 // Owned backing store for variables[].value. snapshot() copies each

149 // Core of eval(): inject the paused frame's variables into the workspace,
