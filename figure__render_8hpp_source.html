46 static Rgb fromHex(const std::string &hex)
47 {
48 if (hex.size() >= 7 && hex[0] == '#') {
49 auto h = [&](size_t i) {
50 return static_cast<std::uint8_t>(std::strtoul(hex.substr(i, 2).c_str(), nullptr, 16));

55 }

61 }

120// One MATLAB colour spec → Rgb. Accepts the linespec short letters,
121// long colour names, #rrggbb, and "r,g,b" / "r g b" float triplets
122// (0..1, MATLAB convention). Returns false when unrecognised.
123inline bool textColorSpec(const std::string &v, Rgb &out)
124{
125 // MATLAB-exact named colours (short letters are the SAME values:
126 // 'r' == 'red' == (1,0,0)) — NOT the soft series palette, which is
127 // a deliberate nk aesthetic for LINES only.
128 static const std::pair<const char *, Rgb> kNames[] = {
129 {"red", {255, 0, 0}}, {"green", {0, 128, 0}},
130 {"blue", {0, 0, 255}}, {"cyan", {0, 255, 255}},
131 {"magenta", {255, 0, 255}}, {"yellow", {255, 255, 0}},
132 {"black", {0, 0, 0}}, {"white", {255, 255, 255}},
133 {"none", {255, 255, 255}},
134 };
135 std::string s;
136 for (char c : v)
137 if (c != ' ' && c != '[' && c != ']')
138 s += (char)std::tolower((unsigned char)c);
139 if (s.size() == 1) {
140 for (const auto &kv : kNames)
141 if (s[0] == kv.first[0]) {
142 out = kv.second;
143 return true;
144 }
145 return false;
146 }
147 for (const auto &kv : kNames)
148 if (s == kv.first) {
149 out = kv.second;
150 return true;
151 }
152 if (s.size() >= 7 && s[0] == '#') {
153 out = Rgb::fromHex(s);
154 return true;
155 }
156 double ch[3] = {0, 0, 0};
157 int n = 0;
158 size_t i = 0;
159 while (i < s.size() && n < 3) {
160 size_t e = i;
161 while (e < s.size() && s[e] != ',')
162 ++e;
163 try {
164 ch[n++] = std::stod(s.substr(i, e - i));
165 } catch (...) {
166 return false;
167 }
168 i = e + 1;
169 }
170 if (n == 3) {
171 out = Rgb((std::uint8_t)std::lround(std::min(1.0, std::max(0.0, ch[0])) * 255),
172 (std::uint8_t)std::lround(std::min(1.0, std::max(0.0, ch[1])) * 255),
173 (std::uint8_t)std::lround(std::min(1.0, std::max(0.0, ch[2])) * 255));
174 return true;
175 }
176 return false;
177}

179inline TextExtras parseTextExtras(const std::string &style)
180{
181 TextExtras t;
182 if (style.empty())
183 return t;
184 size_t i = 0;
185 while (i < style.size()) {
186 size_t semi = style.find(';', i);
187 std::string kv = style.substr(
188 i, (semi == std::string::npos ? style.size() : semi) - i);
189 size_t eq = kv.find('=');
190 if (eq != std::string::npos) {
191 std::string key = kv.substr(0, eq);
192 std::string val = kv.substr(eq + 1);
193 if (key == "color") {
194 Rgb c;
195 if (textColorSpec(val, c))
196 t.color = c;
197 } else if (key == "rgb") {
198 Rgb c;
199 if (textColorSpec(val, c))
200 t.color = c;
201 } else if (key == "fontSize") {
202 try {
203 t.fontSize = std::stod(val);
204 } catch (...) {
205 }
206 } else if (key == "halign") {
207 t.halign = val == "center" ? 1 : val == "right" ? 2 : 0;
208 }
209 }
210 i = semi == std::string::npos ? style.size() : semi + 1;
211 }
212 return t;
213}

215inline StyleParts parseStyle(const std::string &s)
216{
217 StyleParts p;
218 if (s.empty())
219 return p;
220 if (s.find('=') != std::string::npos) {
221 for (size_t i = 0; i < s.size();) {
222 size_t eq = s.find('=', i);
223 if (eq == std::string::npos)
224 break;
225 size_t semi = s.find(';', eq);
226 std::string key = s.substr(i, eq - i);
227 std::string val = s.substr(
228 eq + 1, (semi == std::string::npos ? s.size() : semi) - eq - 1);
229 if (key == "color") {
230 p.color = Rgb::fromHex(val);
231 p.hasColor = true;
232 } else if (key == "lineStyle") {
233 p.lineStyle = val;
234 } else if (key == "marker") {
235 p.marker = val;
236 } else if (key == "faceColor") {
237 p.faceColor = Rgb::fromHex(val);
238 p.hasFaceColor = true;
239 } else if (key == "edgeColor") {
240 p.edgeColor = Rgb::fromHex(val);
241 p.hasEdgeColor = true;
242 } else if (key == "binWidth") {
243 try {
244 p.binWidth = std::stod(val);
245 } catch (...) {
246 }
247 } else if (key == "faceAlpha") {
248 try {
249 p.faceAlpha = std::stod(val);
250 } catch (...) {
251 }
252 }
253 i = semi == std::string::npos ? s.size() : semi + 1;
254 }
255 return p;
256 }
257 static const std::string markers = "o+*.xsd^v<>ph";
258 size_t i = 0;
259 while (i < s.size()) {
260 if (p.lineStyle.empty() && s.compare(i, 2, "--") == 0) { p.lineStyle = "--"; i += 2; continue; }
261 if (p.lineStyle.empty() && s.compare(i, 2, "-.") == 0) { p.lineStyle = "-."; i += 2; continue; }
262 char c = s[i];
263 if (p.lineStyle.empty() && (c == '-' || c == ':')) { p.lineStyle = std::string(1, c); ++i; continue; }
264 if (!p.hasColor) {
265 Rgb col;
266 if (charSpecColor(c, col)) {
267 p.color = col;
268 p.hasColor = true;
269 ++i;
270 continue;
271 }
272 }
273 if (p.marker.empty() && markers.find(c) != std::string::npos) {
274 p.marker = std::string(1, c);
275 ++i;
276 continue;
277 }
278 ++i;
279 }
280 return p;
281}

335// Custom palette from colormap(M) JSON "[[r,g,b],...]" (values [0,1]).
336// figdetail::parseFlatDoubles stops at the first ']' (flat arrays), so
337// brackets are stripped first and the numbers re-wrapped — "[[0,0,0],
338// [1,0,0]]" → "[0,0,0,1,0,0]" — then grouped into RGB triplets.
339inline std::vector<Rgb> parseCustomPalette(const std::string &json)
340{
341 std::string flat = "[";
342 for (char ch : json)
343 if (ch != '[' && ch != ']')
344 flat += ch;
345 flat += "]";
346 auto v = figdetail::parseFlatDoubles(flat);
347 std::vector<Rgb> out;
348 out.reserve(v.size() / 3);
349 for (size_t k = 0; k + 3 <= v.size(); k += 3)
350 out.push_back(Rgb((std::uint8_t)std::lround(v[k] * 255),
351 (std::uint8_t)std::lround(v[k + 1] * 255),
352 (std::uint8_t)std::lround(v[k + 2] * 255)));
353 return out;
354}

393// Screen-space RGB buffer for an IMAGE cmd, shared by the raster and
394// SVG backends so both produce identical pixels. The buffer size is the
395// rounded on-screen rect (at least 1x1); each output pixel samples the
396// containing cell via its centre (MATLAB-style nearest cell). Row/col 0
397// sit at the screen end where the DATA rect's iy0/ix0 edge landed — the
398// axes transform has already baked in yDir/xDir.
399inline std::vector<std::uint8_t> imageCmdRgb(const Cmd &c)
400{
401 double sx0 = std::min(c.xs[0], c.xs[1]), sx1 = std::max(c.xs[0], c.xs[1]);
402 double sy0 = std::min(c.ys[0], c.ys[1]), sy1 = std::max(c.ys[0], c.ys[1]);
403 int pw = std::max(1, (int)std::lround(sx1 - sx0));
404 int ph = std::max(1, (int)std::lround(sy1 - sy0));
405 std::vector<std::uint8_t> buf((size_t)pw * ph * 3, 255);
406
407 // u increases from the screen end of data edge ix0 toward ix1; same
408 // for v with iy0. When the transform flipped the axis, the ix0 end
409 // is the HIGH screen coordinate.
410 auto frac = [](double lo, double hi, double v) {
411 return hi > lo ? (v - lo) / (hi - lo) : 0.5;
412 };
413 std::vector<Rgb> custom;
414 if (!c.cmapCustom.empty())
415 custom = parseCustomPalette(c.cmapCustom);
416 int nPal = 256;
417 const std::uint8_t *pal = c.cmapCustom.empty()
418 ? namedPalette(c.cmap, nPal) : nullptr;
419 if (!custom.empty())
420 nPal = (int)custom.size();
421
422 for (int py = 0; py < ph; ++py) {
423 double cy = sy0 + py + 0.5;
424 double v = c.ys[0] <= c.ys[1] ? frac(sy0, sy1, cy)
425 : 1.0 - frac(sy0, sy1, cy);
426 int row = (int)(v * c.imgH);
427 if (row < 0) row = 0;
428 if (row >= c.imgH) row = c.imgH - 1;
429 for (int px = 0; px < pw; ++px) {
430 double cx = sx0 + px + 0.5;
431 double u = c.xs[0] <= c.xs[1] ? frac(sx0, sx1, cx)
432 : 1.0 - frac(sx0, sx1, cx);
433 int col = (int)(u * c.imgW);
434 if (col < 0) col = 0;
435 if (col >= c.imgW) col = c.imgW - 1;
436 Rgb rgb(255, 255, 255);
437 if (c.rgbDirect) {
438 size_t o = ((size_t)row * c.imgW + col) * 4; // RGBA quad
439 rgb = Rgb(c.img[o], c.img[o + 1], c.img[o + 2]);
440 } else {
441 std::uint8_t idx = c.img[(size_t)col * c.imgH + row];
442 int e = 0; // NaN sentinel and t<=0 both give entry 1
443 if (idx != 255) {
444 double val = c.qLo + (idx / 254.0) * (c.qHi - c.qLo);
445 double t = (val - c.climLo) / (c.climHi - c.climLo);
446 if (t < 0) t = 0;
447 if (t > 1) t = 1;
448 e = (int)std::lround(t * (nPal - 1));
449 if (e < 0) e = 0;
450 if (e >= nPal) e = nPal - 1;
451 }
452 if (pal) {
453 rgb = Rgb(pal[e * 3], pal[e * 3 + 1], pal[e * 3 + 2]);
454 } else {
455 rgb = custom[std::min((size_t)e, custom.size() - 1)];
456 }
457 }
458 size_t o = ((size_t)py * pw + px) * 3;
459 buf[o] = rgb.r;
460 buf[o + 1] = rgb.g;
461 buf[o + 2] = rgb.b;
462 }
463 }
464 return buf;
465}

467// Nested 2-D JSON rows "[[a,b],[c,d]]" → row-major doubles + dims.
468// Null entries (NaN cells) → NaN. Depth-tracked: only INNER brackets
469// (depth 2) delimit rows — the outer pair is the wrapper (counting it
470// as a row over-read the value vector).
471inline void parseNestedDoubles(const std::string &j, std::vector<double> &out,
472 size_t &rows, size_t &cols)
473{
474 out.clear();
475 rows = cols = 0;
476 size_t i = 0;
477 size_t colCount = 0;
478 int depth = 0;
479 while (i < j.size()) {
480 char c = j[i];
481 if (c == '[') {
482 ++depth;
483 if (depth == 2)
484 colCount = 0;
485 ++i;
486 } else if (c == ']') {
487 if (depth == 2) {
488 ++rows;
489 if (cols == 0)
490 cols = colCount;
491 }
492 --depth;
493 ++i;
494 } else if (c == ',' || c == ' ' || c == '\n' || c == '\t') {
495 ++i;
496 } else if (depth >= 2) {
497 size_t e = i;
498 while (e < j.size() && j[e] != ',' && j[e] != ']')
499 ++e;
500 std::string tok = j.substr(i, e - i);
501 bool nullTok = tok.find("null") != std::string::npos;
502 try {
503 out.push_back(nullTok ? std::nan("") : std::stod(tok));
504 } catch (...) {
505 out.push_back(std::nan(""));
506 }
507 ++colCount;
508 i = e;
509 } else {
510 ++i;
511 }
512 }
513 if (!rows)
514 out.clear();
515}

517// ── Ticks (ported from ide/src/components/plot/plotTicks.js) ─────────
518// MATLAB-exact auto ticks (probe-verified R2025b, portion 34): target
519// 10 intervals, ladder breakpoints 1/2/5 — [0,10]→step 1 (11 ticks),
520// [0,3]→step 0.5 (7), [0,7]→step 1 (8).
521inline std::vector<double> niceTicks(double mn, double mx, int target = 10)
522{
523 std::vector<double> major;
524 double range = mx - mn;
525 if (!(range > 0))
526 return {mn};
527 double rough = range / target;
528 double pow = std::pow(10.0, std::floor(std::log10(rough)));
529 double norm = rough / pow;
530 double step = norm <= 1.0 ? pow
531 : norm <= 2.0 ? 2 * pow
532 : norm <= 5.0 ? 5 * pow : 10 * pow;
533 for (double v = std::ceil(mn / step) * step; v <= mx + step * 1e-6; v += step)
534 major.push_back(std::stod(std::to_string(v).substr(0, 12)));
535 return major;
536}

553// Default tick label: %g-style with trailing-zero trim + exponent
554// compaction ("1e+06" → "1e6").
555inline std::string tickLabel(double v)
556{
557 if (v == 0)
558 return "0";

560 std::snprintf(buf, sizeof buf, "%.6g", v);
561 std::string s = buf;
562 size_t e = s.find('e');

564 std::string mant = s.substr(0, e);

566 bool neg = !exp.empty() && exp[0] == '-';
567 size_t k = 0;
568 while (k < exp.size() && (exp[k] == '+' || exp[k] == '-'))
569 ++k;
570 while (k + 1 < exp.size() && exp[k] == '0')
571 exp.erase(k, 1);
572 if (mant.size() > 1 && mant.compare(mant.size() - 1, 1, "0") == 0
573 && mant.find('.') != std::string::npos)
574 while (mant.back() == '0')
575 mant.pop_back();

577 mant.pop_back();
578 s = mant + "e" + (neg ? "-" : "") + exp.substr(k);
579 } else if (s.find('.') != std::string::npos) {
580 while (!s.empty() && s.back() == '0')
581 s.pop_back();
582 if (!s.empty() && s.back() == '.')
583 s.pop_back();
584 }
585 return s;

709{

711 const Rgb kGridCol(0xd8, 0xd8, 0xdc);
712 const Rgb kTextCol(0x28, 0x28, 0x30);

714 // Margin: room for tick labels + axis labels. axis off (`axis off`,
715 // imshow) drops the decorations — a slim inset only, the image gets
716 // the canvas (probe: imshow fills the saved figure).
717 double ml = 64, mr = 14, mt = 20, mb = 48;
718 if (!ax.title.empty())
719 mt += 18;
720 // Colorbar reserves space on its side (MATLAB eastoutside shrinks the
721 // axes to ~0.68 width; we reserve strip + ticks + labels).
722 std::string cbSide = ax.colorbarLocation;
723 for (char &c : cbSide)
724 c = (char)std::tolower((unsigned char)c);
725 if (cbSide.size() > 7
726 && cbSide.compare(cbSide.size() - 7, 7, "outside") == 0)
727 cbSide.erase(cbSide.size() - 7);
728 if (cbSide.empty() || cbSide == "best")
729 cbSide = ax.colorbarLocation.empty() ? "" : "east";
730 if (!cbSide.empty() && cbSide != "none") {
731 if (cbSide == "east")
732 mr += 48;
733 else if (cbSide == "west")
734 ml += 48;
735 else if (cbSide == "south")
736 mb += 48;
737 else if (cbSide == "north")
738 mt += 34;
739 }
740 Rect plot{box.x0 + ml, box.y0 + mt, box.x1 - mr, box.y1 - mb};
741 if (!ax.axisVisible) {
742 plot = Rect{box.x0 + 4, box.y0 + 4, box.x1 - 4, box.y1 - 4};
743 if (!ax.title.empty())
744 plot.y0 += 18;
745 }
746
747 // ── Collect series data → ranges ────────────────────────────────
748 auto applyLim = [](const std::string &json, double &lo, double &hi) {
749 if (json.empty())
750 return;
751 auto v = figdetail::parseFlatDoubles(json);
752 if (v.size() == 2) {
753 lo = v[0];
754 hi = v[1];
755 }
756 };
757 double xlo = +INFINITY, xhi = -INFINITY, ylo = +INFINITY, yhi = -INFINITY;
758 std::vector<std::vector<double>> xs(ax.datasets.size()), ys(ax.datasets.size());
759 int imgDatasets = 0, seriesDatasets = 0;
760
761 // Image data rect (probe-verified MATLAB R2025b, portion 35). The
762 // wire xJson/yJson are "[1,C]"/"[1,R]" ranges by default, a
763 // per-column/row CENTER vector when numel==cells, or EDGE vectors
764 // when numel==cells+1. Image edges: centers get ±spacing/2 (the
765 // default [1,C] reduces to ±0.5); edge vectors are used as-is. When
766 // the axes limits are auto, MATLAB pads the image extent by
767 // span/(2*(cells-1)) where span = x(end)-x(1) of the GIVEN vector —
768 // e.g. default 3x4 image → XLim [0.5 4.5]; centers [10..40] →
769 // [5 45]; edges [10..50] → [3.333 56.667].
770 auto imageEdges = [](const std::vector<double> &v, size_t cells,
771 double &e0, double &e1, double &pad) {
772 if (v.size() < 2 || cells == 0) {
773 e0 = 0.5;
774 e1 = cells > 0 ? cells + 0.5 : 1.5;
775 pad = 0.5;
776 return;
777 }
778 double span = v.back() - v.front();
779 if (v.size() == cells + 1) { // edges as given
780 e0 = v.front();
781 e1 = v.back();
782 } else { // centers (or [1,C] range)
783 double sp = cells > 1 ? span / (double)(cells - 1) : (span > 0 ? span : 1.0);
784 e0 = v.front() - sp / 2;
785 e1 = v.back() + sp / 2;
786 }
787 pad = cells > 1 ? span / (2.0 * (double)(cells - 1)) : (span > 0 ? span / 2 : 0.5);
788 };
789 for (size_t i = 0; i < ax.datasets.size(); ++i) {
790 const DatasetInfo &ds = ax.datasets[i];
791 if (!ds.visible)
792 continue;
793 if (ds.type == "surf" || ds.type == "mesh" || ds.type == "waterfall"
794 || (ds.type == "plot3" && !ds.zJson.empty())) {
795 continue; // rendered in the 3-D block above
796 }
797 if (ds.type == "imagesc" || ds.type == "image-rgb") {
798 ++imgDatasets;
799 size_t cols = ds.originalCols ? ds.originalCols : 1;
800 size_t rows = ds.originalRows ? ds.originalRows : 1;
801 double e0, e1, pd;
802 imageEdges(parseVec(ds.xJson), cols, e0, e1, pd);
803 xlo = std::min(xlo, e0 - pd);
804 xhi = std::max(xhi, e1 + pd);
805 imageEdges(parseVec(ds.yJson), rows, e0, e1, pd);
806 ylo = std::min(ylo, e0 - pd);
807 yhi = std::max(yhi, e1 + pd);
808 continue;
809 }
810 if (ds.type == "text")
811 continue; // deferred: text layer
812 ++seriesDatasets;
813 xs[i] = parseVec(ds.xJson);
814 ys[i] = parseVec(ds.yJson);
815 if (xs[i].empty()) {
816 xs[i].resize(ys[i].size());
817 for (size_t k = 0; k < ys[i].size(); ++k)
818 xs[i][k] = (double)(k + 1);
819 }
820 double a, b;
821 dataRange(xs[i], a, b);
822 xlo = std::min(xlo, a);
823 xhi = std::max(xhi, b);
824 dataRange(ys[i], a, b);
825 ylo = std::min(ylo, a);
826 yhi = std::max(yhi, a * 0 + b);
827 // quiver: arrow tips extend the data range (u/v offsets).
828 if (ds.type == "quiver") {
829 auto uv = parseVec(ds.uJson);
830 auto vv = parseVec(ds.vJson);
831 const size_t nq = std::min(xs[i].size(), ys[i].size());
832 for (size_t k = 0; k < nq && k < uv.size(); ++k) {
833 xlo = std::min(xlo, xs[i][k] + uv[k]);
834 xhi = std::max(xhi, xs[i][k] + uv[k]);
835 }
836 for (size_t k = 0; k < nq && k < vv.size(); ++k) {
837 ylo = std::min(ylo, ys[i][k] + vv[k]);
838 yhi = std::max(yhi, ys[i][k] + vv[k]);
839 }
840 }
841 // errorbar extents
842 auto ext = [&](const std::string &j, bool pos) {
843 auto e = parseVec(j);
844 if (e.empty())
845 return;
846 for (size_t k = 0; k < e.size() && k < ys[i].size(); ++k) {
847 ylo = std::min(ylo, ys[i][k] - (pos ? 0 : e[k]));
848 yhi = std::max(yhi, ys[i][k] + (pos ? e[k] : 0));
849 }
850 };
851 ext(ds.eJson, true);
852 ext(ds.eNegJson, false);
853 ext(ds.ePosJson, true);
854 if (ds.type == "stem") {
855 ylo = std::min(ylo, 0.0);
856 yhi = std::max(yhi, 0.0);
857 }
858 }
859 if (!(xhi > xlo)) {
860 xlo = 0;
861 xhi = xhi > xlo ? xhi : xlo + 1;
862 }
863 if (!(yhi > ylo)) {
864 double mid = ylo;
865 ylo = mid - 0.5;
866 yhi = mid + 0.5;
867 }
868 applyLim(ax.xlimJson, xlo, xhi);
869 applyLim(ax.ylimJson, ylo, yhi);
870 // 5% padding on non-explicit ranges (MATLAB-style breathing room).
871 auto pad = [](double &lo, double &hi, const std::string &lim) {
872 if (!lim.empty())
873 return;
874 double d = (hi - lo) * 0.05;
875 lo -= d;
876 hi += d;
877 };
878 // Image-only axes keep MATLAB's exact image limits (probe: XLim
879 // [0.5 4.5] for a 3x4 — NO extra 5% breathing room). Mixed
880 // image+series axes approximate MATLAB's union algorithm with the
881 // usual 5% pad on top (rare in practice; noted in memory).
882 bool imageOnly = imgDatasets > 0 && seriesDatasets == 0;
883 if (!imageOnly) {
884 pad(xlo, xhi, ax.xlimJson);
885 pad(ylo, yhi, ax.ylimJson);
886 }
887 if (ax.xscale == "log") {
888 xlo = std::max(xlo, 1e-12);
889 if (!(xhi > xlo))
890 xhi = xlo * 100;
891 }
892 if (ax.yscale == "log") {
893 ylo = std::max(ylo, 1e-12);
894 if (!(yhi > ylo))
895 yhi = ylo * 100;
896 }
897
898 // axis image / axis equal — equal data units per pixel on both
899 // axes: shrink the plot rect to the data aspect (MATLAB keeps the
900 // box centered-ish; probe: imshow renders undistorted cells).
901 if (ax.axisMode == "image" || ax.axisMode == "equal") {
902 double dataAspect = (xhi - xlo) / (yhi - ylo);
903 double pxAspect = (plot.x1 - plot.x0) / (plot.y1 - plot.y0);
904 if (dataAspect > 0 && pxAspect > 0) {
905 if (pxAspect > dataAspect) {
906 double w = (plot.y1 - plot.y0) * dataAspect;
907 double cx = (plot.x0 + plot.x1) / 2;
908 plot.x0 = cx - w / 2;
909 plot.x1 = cx + w / 2;
910 } else {
911 double h = (plot.x1 - plot.x0) / dataAspect;
912 double cy = (plot.y0 + plot.y1) / 2;
913 plot.y0 = cy - h / 2;
914 plot.y1 = cy + h / 2;
915 }
916 }
917 }
918
919 AxisRange XR{xlo, xhi, ax.xscale == "log"};
920 AxisRange YR{ylo, yhi, ax.yscale == "log"};
921 double px0 = plot.x0, px1 = plot.x1;
922 double py0 = plot.y1, py1 = plot.y0; // y DOWN: data-up = pixel-up
923 if (ax.xDir == "reverse")
924 std::swap(px0, px1);
925 if (ax.yDir == "reverse")
926 std::swap(py0, py1);
927 auto X = [&](double v) { return XR.map(v, px0, px1); };
928 auto Y = [&](double v) { return YR.map(v, py0, py1); };
929
930 // ── Grid (under the data) ───────────────────────────────────────
931 if (ax.gridMajor && ax.gridUserTouched) {
932 auto gx = XR.log ? logTicks(xlo, xhi) : niceTicks(xlo, xhi);
933 auto gy = YR.log ? logTicks(ylo, yhi) : niceTicks(ylo, yhi);
934 for (double v : gx)
935 addLine(out, {X(v), X(v)}, {plot.y0, plot.y1}, kGridCol, 1, false);
936 for (double v : gy)
937 addLine(out, {plot.x0, plot.x1}, {Y(v), Y(v)}, kGridCol, 1, false);
938 }
939
940 // ── Images (under series — MATLAB draws lines over the image) ───
941 for (size_t i = 0; i < ax.datasets.size(); ++i) {
942 const DatasetInfo &ds = ax.datasets[i];
943 if (!ds.visible || (ds.type != "imagesc" && ds.type != "image-rgb"))
944 continue;
945 size_t cols = ds.originalCols ? ds.originalCols : 1;
946 size_t rows = ds.originalRows ? ds.originalRows : 1;
947 double e0, e1, pd;
948 imageEdges(parseVec(ds.xJson), cols, e0, e1, pd);
949 double xe0 = e0, xe1 = e1;
950 imageEdges(parseVec(ds.yJson), rows, e0, e1, pd);
951 double ye0 = e0, ye1 = e1;
952 Cmd c;
953 c.kind = Cmd::IMAGE;
954 if (ds.type == "imagesc") {
955 c.img = ds.zQuantized;
956 c.imgW = (int)cols;
957 c.imgH = (int)rows;
958 c.qLo = ds.cminOrig;
959 c.qHi = ds.cmaxOrig;
960 auto cl = figdetail::parseFlatDoubles(ax.climJson);
961 if (cl.size() == 2 && cl[0] < cl[1]) {
962 c.climLo = cl[0];
963 c.climHi = cl[1];
964 } else {
965 c.climLo = c.qLo; // CLim auto = data range
966 c.climHi = c.qHi;
967 }
968 c.cmap = ax.colormapName.empty() ? "parula" : ax.colormapName;
969 c.cmapCustom = ax.customColormapJson;
970 } else {
971 c.img = ds.rgbBytes;
972 c.imgW = (int)cols;
973 c.imgH = (int)rows;
974 c.rgbDirect = true;
975 }
976 c.ix0 = xe0;
977 c.ix1 = xe1;
978 c.iy0 = ye0;
979 c.iy1 = ye1;
980 c.xs = {X(xe0), X(xe1)};
981 c.ys = {Y(ye0), Y(ye1)};
982 out.push_back(std::move(c));
983 }

985 // ── 3-D content (portion 40; MATLAB probes R2025b): surf = flat ──
986 // per-quad faces coloured by the FIRST corner's Z through the
987 // colormap (FaceColor 'flat', CData=Z), edges (0.129)x3; mesh /
988 // waterfall = segments coloured by Z (EdgeColor 'flat'); view from
989 // ax.viewJson, default (-37.5, 30); orthographic az/el projection;
990 // painter's sort (far quads first). clim auto = [min Z, max Z] —
991 // same rule as imagesc. The 3-D box replaces the 2-D frame.
992 {
993 bool has3d = false;
994 for (const auto &ds : ax.datasets)
995 if (ds.visible
996 && (ds.type == "surf" || ds.type == "mesh"
997 || ds.type == "waterfall"
998 || (ds.type == "plot3" && !ds.zJson.empty()))
999 && !ds.zJson.empty())
1000 has3d = true;
1001 if (has3d) {
1002 // Bounds over all 3-D datasets.
1003 double xlo3 = 1e300, xhi3 = -1e300, ylo3 = 1e300,
1004 yhi3 = -1e300, zlo3 = 1e300, zhi3 = -1e300;
1005 auto acc = [](double &lo, double &hi, double v) {
1006 if (!std::isfinite(v))
1007 return;
1008 lo = std::min(lo, v);
1009 hi = std::max(hi, v);
1010 };
1011 struct SurfGrid {
1012 std::vector<double> z; // row-major R x C
1013 std::vector<double> gx; // column coords
1014 std::vector<double> gy; // row coords
1015 size_t r, c;
1016 };
1017 std::vector<SurfGrid> grids;
1018 for (const auto &ds : ax.datasets) {
1019 if (!ds.visible || ds.zJson.empty())
1020 continue;
1021 if (ds.type == "surf" || ds.type == "waterfall") {
1022 auto xv = parseVec(ds.xJson);
1023 auto yv = parseVec(ds.yJson);
1024 std::vector<double> zf;
1025 size_t r = 0, c = 0;
1026 parseNestedDoubles(ds.zJson, zf, r, c);
1027 if (xv.empty() || yv.empty() || !r || !c)
1028 continue;
1029 SurfGrid g;
1030 g.z = std::move(zf);
1031 for (size_t cc = 0; cc < c; ++cc)
1032 g.gx.push_back(xv[std::min(cc, xv.size() - 1)]);
1033 for (size_t rr = 0; rr < r; ++rr)
1034 g.gy.push_back(yv[std::min(rr, yv.size() - 1)]);
1035 g.r = r;
1036 g.c = c;
1037 for (size_t rr = 0; rr < r; ++rr)
1038 for (size_t cc = 0; cc < c; ++cc) {
1039 acc(xlo3, xhi3, g.gx[cc]);
1040 acc(ylo3, yhi3, g.gy[rr]);
1041 acc(zlo3, zhi3, g.z[rr * c + cc]);
1042 }
1043 grids.push_back(std::move(g));
1044 } else if (ds.type == "mesh" || ds.type == "plot3") {
1045 auto xv = parseVec(ds.xJson);
1046 auto yv = parseVec(ds.yJson);
1047 auto zv = parseVec(ds.zJson);
1048 size_t n = std::min(std::min(xv.size(), yv.size()),
1049 zv.size());
1050 for (size_t k = 0; k < n; ++k) {
1051 acc(xlo3, xhi3, xv[k]);
1052 acc(ylo3, yhi3, yv[k]);
1053 acc(zlo3, zhi3, zv[k]);
1054 }
1055 }
1056 }
1057 if (!(xhi3 > xlo3)) { xlo3 = 0; xhi3 = 1; }
1058 if (!(yhi3 > ylo3)) { ylo3 = 0; yhi3 = 1; }
1059 if (!(zhi3 > zlo3)) { zlo3 = 0; zhi3 = 1; }
1060 applyLim(ax.xlimJson, xlo3, xhi3);
1061 applyLim(ax.ylimJson, ylo3, yhi3);
1062 applyLim(ax.zlimJson, zlo3, zhi3);
1063 // CLim + palette (same resolution as imagesc/colorbar).
1064 double clo3 = zlo3, chi3 = zhi3;
1065 {
1066 auto cl = figdetail::parseFlatDoubles(ax.climJson);
1067 if (cl.size() == 2 && cl[0] < cl[1]) {
1068 clo3 = cl[0];
1069 chi3 = cl[1];
1070 }
1071 }
1072 std::vector<Rgb> custom3;
1073 if (!ax.customColormapJson.empty())
1074 custom3 = parseCustomPalette(ax.customColormapJson);
1075 int nPal3 = 256;
1076 const std::uint8_t *pal3 =
1077 custom3.empty()
1078 ? namedPalette(ax.colormapName.empty()
1079 ? std::string("parula")
1080 : ax.colormapName, nPal3)
1081 : nullptr;
1082 if (!custom3.empty())
1083 nPal3 = (int)custom3.size();
1084 auto zColor = [&](double z) -> Rgb {
1085 if (!std::isfinite(z))
1086 z = clo3;
1087 double t = (z - clo3) / (chi3 - clo3);
1088 if (t < 0) t = 0;
1089 if (t > 1) t = 1;
1090 int e = (int)std::lround(t * (nPal3 - 1));
1091 if (e < 0) e = 0;
1092 if (e >= nPal3) e = nPal3 - 1;
1093 return pal3 ? Rgb(pal3[e * 3], pal3[e * 3 + 1],
1094 pal3[e * 3 + 2])
1095 : custom3[(size_t)e];
1096 };
1097 // View (az, el) in degrees; MATLAB orthographic transform.
1098 double azDeg = -37.5, elDeg = 30;
1099 {
1100 auto vv = figdetail::parseFlatDoubles(ax.viewJson);
1101 if (vv.size() == 2) {
1102 azDeg = vv[0];
1103 elDeg = vv[1];
1104 }
1105 }
1106 const double az = azDeg * 3.14159265358979323846 / 180.0;
1107 const double el = elDeg * 3.14159265358979323846 / 180.0;
1108 const double sAz = std::sin(az), cAz = std::cos(az);
1109 const double sEl = std::sin(el), cEl = std::cos(el);
1110 auto norm3 = [&](double v, double lo, double hi) {
1111 return hi > lo ? (v - lo) / (hi - lo) - 0.5 : 0.0;
1112 };
1113 // (sx, sy) in MATLAB screen space (y up), empirically
1114 // pinned by the corner-blob orientation test (view [0 90],
1115 // hot corner at data (x_min, y_min) renders LEFT-BOTTOM in
1116 // MATLAB R2025b): sx = -x*sin + y*cos; sy = (x*cos +
1117 // y*sin)*sin(el) + z*cos(el). Depth toward the camera =
1118 // -forward . p with forward = right x up (painter's sort,
1119 // larger = nearer).
1120 auto proj = [&](double x, double y, double z,
1121 double &sx, double &sy, double &depth) {
1122 double xn = norm3(x, xlo3, xhi3);
1123 double yn = norm3(y, ylo3, yhi3);
1124 double zn = norm3(z, zlo3, zhi3);
1125 sx = -xn * sAz + yn * cAz;
1126 sy = (xn * cAz + yn * sAz) * sEl + zn * cEl;
1127 depth = zn * sEl - cEl * (xn * cAz + yn * sAz);
1128 };
1129 // Fit the projected unit cube into the plot rect.
1130 double minSx = 1e300, maxSx = -1e300, minSy = 1e300,
1131 maxSy = -1e300;
1132 for (int corner = 0; corner < 8; ++corner) {
1133 double sx, sy, d;
1134 proj(corner & 1 ? xhi3 : xlo3, corner & 2 ? yhi3 : ylo3,
1135 corner & 4 ? zhi3 : zlo3, sx, sy, d);
1136 minSx = std::min(minSx, sx);
1137 maxSx = std::max(maxSx, sx);
1138 minSy = std::min(minSy, sy);
1139 maxSy = std::max(maxSy, sy);
1140 }
1141 double spanX = std::max(1e-9, maxSx - minSx);
1142 double spanY = std::max(1e-9, maxSy - minSy);
1143 double availW = (plot.x1 - plot.x0) * 0.92;
1144 double availH = (plot.y1 - plot.y0) * 0.92;
1145 double sc = std::min(availW / spanX, availH / spanY);
1146 auto PX = [&](double sx) {
1147 return (plot.x0 + plot.x1) / 2
1148 + (sx - (minSx + maxSx) / 2) * sc;
1149 };
1150 auto PY = [&](double sy) {
1151 return (plot.y0 + plot.y1) / 2
1152 - (sy - (minSy + maxSy) / 2) * sc;
1153 };
1154 // Box: the 12 cube edges, projected.
1155 if (ax.axisVisible) {
1156 const double bx[2] = {xlo3, xhi3}, by[2] = {ylo3, yhi3},
1157 bz[2] = {zlo3, zhi3};
1158 for (int axis = 0; axis < 3; ++axis)
1159 for (int f1 = 0; f1 <= 1; ++f1)
1160 for (int f2 = 0; f2 <= 1; ++f2) {
1161 double q0[3], q1[3];
1162 int o1 = (axis + 1) % 3, o2 = (axis + 2) % 3;
1163 double fd0[3] = {0, 0, 0}, fd1[3] = {0, 0, 0};
1164 fd0[axis] = 0; fd1[axis] = 1;
1165 fd0[o1] = f1; fd1[o1] = f1;
1166 fd0[o2] = f2; fd1[o2] = f2;
1167 double w0[3] = {bx[(int)fd0[0]], by[(int)fd0[1]],
1168 bz[(int)fd0[2]]};
1169 double w1[3] = {bx[(int)fd1[0]], by[(int)fd1[1]],
1170 bz[(int)fd1[2]]};

1735// whole canvas.
1736inline std::vector<Cmd> renderFigureDisplay(const FigureState &fig, int W,
1737 int H)
1738{

1741 int cols = std::max(1, fig.subplotCols);
1742 // Order axes by subplotIndex (1-based reading order) when gridded.
1743 std::vector<const AxesState *> ordered;
1744 for (int p = 1; p <= rows * cols; ++p)
1745 for (const auto &ax : fig.axes)
1746 if (ax.subplotIndex == p) {
1747 ordered.push_back(&ax);
1748 break;
1749 }
1750 if (ordered.empty())
1751 for (const auto &ax : fig.axes)
1752 ordered.push_back(&ax);
1753 for (size_t i = 0; i < ordered.size(); ++i) {
1754 int r = rows > 1 || cols > 1 ? (int)i / cols : 0;
1755 int c = rows > 1 || cols > 1 ? (int)i % cols : 0;
1756 Rect box;
1757 box.x0 = 10 + c * (W - 20.0) / cols + 6;
1758 box.x1 = 10 + (c + 1) * (W - 20.0) / cols - 6;
1759 box.y0 = 10 + r * (H - 20.0) / rows + 6;
1760 box.y1 = 10 + (r + 1) * (H - 20.0) / rows - 6;
1761 if (fig.subplotRows == 0) {
1762 box = Rect{10, 10, (double)W - 10, (double)H - 10};
1763 }
1764 if (!fig.superTitle.empty() && r == 0 && c == (cols - 1) / 2) {
1765 addText(out, fig.superTitle,
1766 W / 2.0 - 3.4 * fig.superTitle.size(), 6, 13,
1767 Rgb(0x28, 0x28, 0x30));
1768 box.y0 += 10;
1769 }
1770 renderAxes(fig, *ordered[i], box, out);
1771 }
1772 // ── Figure-level annotations (portion 41): normalized [0,1] figure
1773 // units mapped straight onto the canvas. MATLAB defaults: Color
1774 // (0.1294)^3, HeadSize 10 (chevron ~12% of length), textarrow text
1775 // at the TAIL.
1776 for (const auto &an : fig.annotations) {
1777 Rgb col = an.colorHex.empty() ? Rgb(33, 33, 33)
1778 : Rgb::fromHex(an.colorHex);
1779 double ax0 = an.x1 * W, ay0 = (1.0 - an.y1) * H;
1780 double ax1 = an.x2 * W, ay1 = (1.0 - an.y2) * H;
1781 if (an.type == "line" || an.type == "arrow"
1782 || an.type == "textarrow") {
1783 addLine(out, {ax0, ax1}, {ay0, ay1}, col, 0.75, false);
1784 if (an.type != "line") {
1785 double dx = ax1 - ax0, dy = ay1 - ay0;
1786 double len = std::hypot(dx, dy);
1787 if (len > 1e-9) {
1788 double hl = std::max(4.0, len * 0.12);
1789 double ux = dx / len, uy = dy / len;
1790 double a1 = 0.45;
1791 double c1 = std::cos(a1), s1 = std::sin(a1);
1792 addLine(out,
1793 {ax1 - hl * (ux * c1 - uy * s1), ax1},
1794 {ay1 - hl * (uy * c1 + ux * s1), ay1},
1795 col, 0.75, false);
1796 addLine(out,
1797 {ax1 - hl * (ux * c1 + uy * s1), ax1},
1798 {ay1 - hl * (uy * c1 - ux * s1), ay1},
1799 col, 0.75, false);
1800 }
1801 }
1802 if (an.type == "textarrow" && !an.text.empty()) {
1803 double w = 6.0 * (an.fontSize / 10.0) * an.text.size();
1804 addText(out, an.text, ax0 - w - 6, ay0 - 3.5,
1805 an.fontSize, col);
1806 }
1807 } else if (an.type == "rectangle" || an.type == "textbox") {
1808 double x0p = ax0, y0p = ay1, x1p = ax1, y1p = ay0;
1809 if (!an.faceHex.empty()) {
1810 Rgb fc = Rgb::fromHex(an.faceHex);
1811 addPoly(out, {x0p, x1p, x1p, x0p}, {y0p, y0p, y1p, y1p},
1812 fc, true);
1813 }
1814 addLine(out, {x0p, x1p, x1p, x0p, x0p},
1815 {y0p, y0p, y1p, y1p, y0p}, col, 0.75, false);
1816 if (an.type == "textbox" && !an.text.empty())
1817 addText(out, an.text, x0p + 6, (y0p + y1p) / 2 - 3.5,
1818 an.fontSize, Rgb(0x28, 0x28, 0x30));
1819 } else if (an.type == "ellipse") {
1820 double cx = (ax0 + ax1) / 2, cy = (ay0 + ay1) / 2;
1821 double rx = std::abs(ax1 - ax0) / 2, ry = std::abs(ay1 - ay0) / 2;
1822 std::vector<double> ex, ey;
1823 for (int k = 0; k <= 64; ++k) {
1824 double t = 2 * 3.14159265358979323846 * k / 64.0;
1825 ex.push_back(cx + rx * std::cos(t));
1826 ey.push_back(cy + ry * std::sin(t));
1827 }
1828 if (!an.faceHex.empty()) {
1829 Rgb fc = Rgb::fromHex(an.faceHex);
1830 addPoly(out, ex, ey, fc, true);
1831 }
1832 for (size_t k = 0; k + 1 < ex.size(); ++k)
1833 addLine(out, {ex[k], ex[k + 1]}, {ey[k], ey[k + 1]}, col,
1834 0.75, false);
1835 }
1836 }
1837 return out;
1838}

1918 + std::to_string(c.y + r) + " M"
