20// Base for "chunk exceeds a bytecode operand-space limit" errors. The engine
21// catches this type to run the script on the TreeWalker instead (the
22// documented dual-engine fallback contract) unless a debug observer needs the
23// VM. Derives from runtime_error so existing broad `catch (std::exception&)`
24// TW-fallbacks (class methods) keep working.
25struct CompilerLimitError : std::runtime_error {
26 using std::runtime_error::runtime_error;
27};
28
29// Thrown when a single chunk needs more than the 255 register slots the
30// bytecode operand byte can address. Distinct type (not a bare runtime_error)
31// so callers can tell "function too large for the VM" apart from other compile
32// failures and surface it instead of silently dropping VM compilation.
33struct RegisterExhaustionError : CompilerLimitError {
34 using CompilerLimitError::CompilerLimitError;
35};

37// Thrown when the constant/string pool index or a jump offset would exceed
38// the int16 operand field and silently wrap (>= 32768 pool entries / > 32767
39// instruction span). Same fallback contract as RegisterExhaustionError.
40struct ChunkOverflowError : CompilerLimitError {
41 using CompilerLimitError::CompilerLimitError;

187 // ── Variable register access API ────────────────────────────
188 //
189 // Every assignment-like compile site MUST call `varRegWrite(name)` so
190 // BytecodeChunk::assignedVars is populated — the debug workspace uses
191 // it to distinguish "user assigned" from "just read" (MATLAB's
192 // whos-parity for shadowed built-ins). Reads and pure re-lookups use
193 // the other two methods.
194 //
195 // varRegWrite(name) — allocates + marks as assigned. Use for every
196 // AST node that writes to `name`.
197 // varRegRead(name) — allocates + emits ASSERT_DEF. Use for checked
198 // reads at the AST level.
199 // varRegLookup(name) — allocates without marking or asserting. Use
200 // for internal plumbing: re-fetching a register
201 // for an already-allocated variable, pre-loading
202 // pseudo-vars (nargin/nargout), workspace-env
203 // imports, etc. NEVER use at a user-assignment
204 // site — that would defeat the whole point.
205 //
206 // `markAssigned(name)` is a rare helper for write sites that bypass
207 // varRegWrite (e.g. move-elimination in compileAssign that writes via
208 // `varRegisters_[name] = src` directly).
209 uint8_t varRegWrite(const std::string &name);
210 uint8_t varRegRead(const std::string &name);
211 // preloadReserved: when first allocating a register for a reserved name
212 // (pi/eps/i/j/…), emit a LOAD_CONST of the engine's value. Correct for a
213 // *read* of the constant; must be false when allocating for a parameter
214 // or assignment target named like a constant (e.g. a param `i`), whose
215 // value comes from the call/assignment — otherwise the pre-load clobbers
216 // the argument with the imaginary unit.
217 uint8_t varRegLookup(const std::string &name, bool preloadReserved = true);
218 void markAssigned(const std::string &name) { chunk_.assignedVars.insert(name); }
