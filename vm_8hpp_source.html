43 // Re-entrant call from C++ INTO a compiled chunk, returning all outputs.
44 // Used when a builtin/hook running mid-dispatch must run user code on the
45 // VM (function handles in cellfun/arrayfun, classdef methods/ctors invoked
46 // from C++) so the body runs on the VM rather than the TreeWalker. Saves
47 // the outer VM state, runs `chunk` as a fresh top-level frame to
48 // completion, harvests its return values, and restores the outer so the
49 // outer dispatch loop resumes seamlessly. ownerClass/isCtor set the new
50 // frame's class context; ctorSeed (constructor case) pre-binds the output
51 // variable. Breakpoints inside the callee fire; a debugger *pause* cannot
52 // suspend across the C++ boundary and surfaces as DebugStopException (same
53 // contract as the legacy execute()). See vm_callbacks_plan.md (P3).
54 std::vector<Value> callReentrant(const BytecodeChunk &chunk, Span<const Value> args,
55 size_t nargout,
56 const std::string &ownerClass = std::string(),
57 bool isCtor = false, const Value *ctorSeed = nullptr);

59 // Schedule a user-code function-handle callback as a pausable VM frame on
60 // behalf of a native continuation (state-machine callbacks — see
61 // callback_builtin.hpp). Resolves the handle (plain or closure cell, with
62 // captures appended) to its compiled chunk, pushes a frame, and attaches
63 // `cont` so the frame's return routes back to cont->step(). Returns false if
64 // the handle is not user code (caller should not have scheduled it). The
65 // caller (a VmContinuation::step) returns true after this so the dispatch
66 // loop runs the new frame.
67 bool pushCallbackFrame(const Value &handle, Span<const Value> args, size_t nargout,
68 std::shared_ptr<VmContinuation> cont);

91 // Unified lookup — tries the script-local bucket first, then
92 // workspace. Returns nullptr when the name isn't a compiled
93 // function (caller falls through to built-ins / external funcs).
94 const BytecodeChunk *findCompiledFunc(const std::string &name) const
95 {
96 if (scriptLocalFuncs_) {
97 auto it = scriptLocalFuncs_->find(name);
98 if (it != scriptLocalFuncs_->end())
99 return &it->second;
100 }
101 if (compiledFuncs_) {
102 auto it = compiledFuncs_->find(name);
103 if (it != compiledFuncs_->end())
104 return &it->second;
105 }
106 return nullptr;
107 }
