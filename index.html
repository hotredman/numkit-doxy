Quick Code Example

#include <nk/bundle/standard_library.hpp>
#include <nk/builtin/builtin.hpp>
#include <nk/signal/signal.hpp>
#include <nk/linalg/linalg.hpp>
#include <iostream>
using namespace nk;
int main() {
// 1. Pure C++ Public API with SIMD acceleration
Value t = builtin::linspace(0.0, 1.0, 1000);
Value sig = builtin::sin(2.0 * 3.141592653589793 * 50.0 * t);
// 2. DSP Signal Processing
Value filtered = signal::filter(signal::fir1(16, 0.2), Value::scalar(1.0), sig);
// 3. Embedded Scripting Engine
auto engine = makeStandardEngine();
engine->eval("x = [1, 2, 3; 4, 5, 6];");
Value res = engine->eval("y = sum(x, 1) * 2;");
std::cout << "Result: " << res.toString() << std::endl;
return 0;
}
